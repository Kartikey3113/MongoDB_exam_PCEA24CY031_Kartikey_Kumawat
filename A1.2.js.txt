// Step 2: Employees earning more than 60000 using $gt
print("--- Step 2 ---");
printjson(db.employees.find({ salary: { $gt: 60000 } }).toArray());

// Step 3: Employees earning between 45000 and 60000 (inclusive) using $gte and $lte
print("--- Step 3 ---");
printjson(db.employees.find({ salary: { $gte: 45000, $lte: 60000 } }).toArray());

// Step 4: Employees of IT or Finance department using $in
print("--- Step 4 ---");
printjson(db.employees.find({ dept: { $in: ["IT", "Finance"] } }).toArray());

// Step 5: Chennai employees earning > 50000 ($and) AND HR or salary > 70000 ($or)
print("--- Step 5 ($and) ---");
printjson(db.employees.find({
  $and: [
    { city: "Chennai" },
    { salary: { $gt: 50000 } }
  ]
}).toArray());

print("--- Step 5 ($or) ---");
printjson(db.employees.find({
  $or: [
    { dept: "HR" },
    { salary: { $gt: 70000 } }
  ]
}).toArray());

// Step 6: Employees not living in Chennai or Bengaluru using $nin
print("--- Step 6 ---");
printjson(db.employees.find({ city: { $nin: ["Chennai", "Bengaluru"] } }).toArray());

// Step 7: Employees whose name starts with A, B, or C using $regex
print("--- Step 7 ---");
printjson(db.employees.find({ name: { $regex: "^[ABC]" } }).toArray());

// Step 8: Show only names & salaries, highest first, skip 2, limit 3
print("--- Step 8 ---");
printjson(
  db.employees.find({}, { _id: 0, name: 1, salary: 1 })
    .sort({ salary: -1 })
    .skip(2)
    .limit(3)
    .toArray()
);
