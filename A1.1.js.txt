// Step 1: Insert one student
print("--- Step 1 ---");
printjson(db.students.insertOne({
  rollNo: 101,
  name: "Arun",
  dept: "CSE",
  marks: 78,
  city: "Chennai"
}));

// Step 2: Insert four more students
print("--- Step 2 ---");
printjson(db.students.insertMany([
  { rollNo: 102, name: "Divya", dept: "ECE", marks: 85, city: "Madurai" },
  { rollNo: 103, name: "Karthik", dept: "CSE", marks: 62, city: "Chennai" },
  { rollNo: 104, name: "Meena", dept: "EEE", marks: 74, city: "Coimbatore" },
  { rollNo: 105, name: "Suresh", dept: "CSE", marks: 91, city: "Trichy" }
]));

// Step 3: Display all students, then display only CSE students
print("--- Step 3 (All Students) ---");
printjson(db.students.find().toArray());

print("--- Step 3 (CSE Students) ---");
printjson(db.students.find({ dept: "CSE" }).toArray());

// Step 4: Correct marks of rollNo 103 to 68
print("--- Step 4 ---");
printjson(db.students.updateOne(
  { rollNo: 103 },
  { $set: { marks: 68 } }
));

// Step 5: Add field semester: 3 to every CSE student
print("--- Step 5 ---");
printjson(db.students.updateMany(
  { dept: "CSE" },
  { $set: { semester: 3 } }
));

// Step 6: Delete record of rollNo 104
print("--- Step 6 ---");
printjson(db.students.deleteOne({ rollNo: 104 }));

// Step 7: Display total remaining students
print("--- Step 7 ---");
print(db.students.countDocuments());
